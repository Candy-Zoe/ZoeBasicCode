继承

继承允许我们定义一个类继承另一个类的属性和方法。

基本继承

Python 
class Animal:
    def __init__(self, name):
        self.name = name
    
    def speak(self):
        print(f"{self.name} 发出声音")

class Dog(Animal):
    def speak(self):
        print(f"{self.name} 汪汪叫")

class Cat(Animal):
    def speak(self):
        print(f"{self.name} 喵喵叫")

# 创建对象
dog = Dog("旺财")
cat = Cat("咪咪")
dog.speak()  # 输出: 旺财 汪汪叫
cat.speak()  # 输出: 咪咪 喵喵叫
旺财 汪汪叫
咪咪 喵喵叫

super() 调用父类方法

Python 
class Person:
    def __init__(self, name, age):
        self.name = name
        self.age = age
    
    def info(self):
        print(f"姓名: {self.name}, 年龄: {self.age}")

class Student(Person):
    def __init__(self, name, age, student_id):
        super().__init__(name, age)  # 调用父类构造方法
        self.student_id = student_id
    
    def info(self):
        super().info()  # 调用父类方法
        print(f"学号: {self.student_id}")

s = Student("张三", 18, "2023001")
s.info()
姓名: 张三, 年龄: 18
学号: 2023001

多重继承

Python 支持多重继承，一个类可以继承多个父类。

基本多重继承

Python 
class Flyable:
    def fly(self):
        print("我可以飞")

class Swimmable:
    def swim(self):
        print("我可以游泳")

class Duck(Animal, Flyable, Swimmable):
    pass

duck = Duck("唐老鸭")
duck.speak()  # 继承自 Animal
duck.fly()    # 继承自 Flyable
duck.swim()   # 继承自 Swimmable
唐老鸭 发出声音
我可以飞
我可以游泳

MRO 方法解析顺序

Python 
class A:
    def method(self):
        print("A")

class B(A):
    def method(self):
        print("B")

class C(A):
    def method(self):
        print("C")

class D(B, C):
    pass

# 查看 MRO
print(D.__mro__)

d = D()
d.method()  # 按照 MRO 顺序调用
⚠️ 注意 多重继承可能带来方法解析顺序复杂的问题，使用时需谨慎。

多态

多态允许不同类的对象对同一消息做出不同的响应。

Python 
class Dog:

        print("汪汪叫")

class Cat:
    def speak(self):
        print("喵喵叫")

抽象类

抽象类不能直接实例化，必须被子类继承并实现抽象方法。

Python 
from abc import ABC, abstractmethod

class Shape(ABC):
    @abstractmethod
    def area(self):
        pass
    
    @abstractmethod
    def perimeter(self):
        pass

class Rectangle(Shape):
    def __init__(self, width, height):
        self.width = width
        self.height = height
    
    def area(self):
        return self.width * self.height
    
    def perimeter(self):
        return 2 * (self.width + self.height)

class Circle(Shape):
    def __init__(self, radius):
        self.radius = radius
    
    def area(self):
        return 3.14159 * self.radius ** 2
    
    def perimeter(self):
        return 2 * 3.14159 * self.radius

# 不能直接实例化抽象类
# shape = Shape()  # 错误！

rect = Rectangle(4, 5)
print(f"矩形面积: {rect.area()}")
矩形面积: 20

示例

Python 
class Point:
    def __init__(self, x, y):
        self.x = x
        self.y = y
    
    def __str__(self):
        return f"Point({self.x}, {self.y})"
    
    def __repr__(self):
        return f"Point({self.x}, {self.y})"
    
    def __add__(self, other):
        return Point(self.x + other.x, self.y + other.y)
    
    def __eq__(self, other):
        return self.x == other.x and self.y == other.y

p1 = Point(1, 2)
p2 = Point(3, 4)
p3 = Point(1, 2)

print(str(p1))           # Point(1, 2)
print(p1 + p2)           # Point(4, 6)
print(p1 == p3)          # True

属性装饰器 @property

使用 @property 将方法转换为属性访问。

Python 
class Circle:
    def __init__(self, radius):
        self._radius = radius  # 私有属性
    
    @property
    def radius(self):
        """获取半径"""
        return self._radius
    
    @radius.setter
    def radius(self, value):
        """设置半径（带验证）"""
        if value < 0:
            raise ValueError("半径不能为负数")
        self._radius = value
    
    @property
    def area(self):
        """计算面积（只读属性）"""
        return 3.14159 * self._radius ** 2

c = Circle(5)
print(f"半径: {c.radius}")
print(f"面积: {c.area}")

c.radius = 10
print(f"新半径: {c.radius}")
print(f"新面积: {c.area}")

生成器

生成器是一种特殊的迭代器，用 yield 语句产生值。

生成器函数

Python 
def countdown(n):
    while n > 0:
        yield n
        n -= 1

for num in countdown(5):
    print(num, end=" ")
print()

# 创建生成器对象
gen = countdown(3)
print(next(gen))  # 3
print(next(gen))  # 2
print(next(gen))  # 1
5 4 3 2 1
3
2
1

生成器表达式

Python 
# 列表推导式（立即生成所有元素）
squares = [x**2 for x in range(5)]

# 生成器表达式（惰性求值）
gen = (x**2 for x in range(5))
print(list(gen))

装饰器

装饰器可以在不修改原函数的情况下增强函数功能。

基本装饰器

Python 
def my_decorator(func):
    def wrapper(*args, **kwargs):
        print("装饰器：函数执行前")
        result = func(*args, **kwargs)
        print("装饰器：函数执行后")
        return result
    return wrapper

@my_decorator
def say_hello(name):
    print(f"你好，{name}！")

say_hello("张三")
装饰器：函数执行前
你好，张三！
装饰器：函数执行后

带参数的装饰器

Python 
import time

def timer(seconds):
    def decorator(func):
        def wrapper(*args, **kwargs):
            start = time.time()
            result = func(*args, **kwargs)
            elapsed = time.time() - start
            print(f"执行时间: {elapsed:.2f}秒")
            return result
        return wrapper
    return decorator

@timer(2)
def slow_function():
    time.sleep(0.1)
    return "完成"

result = slow_function()
print(result)

functools.wraps 保留原函数信息

Python 
from functools import wraps

def my_decorator(func):
    @wraps(func)  # 保留原函数信息
    def wrapper(*args, **kwargs):
        return func(*args, **kwargs)
    return wrapper

@my_decorator
def original_func():
    """这是原始函数的文档字符串"""
    pass

print(original_func.__name__)    # original_func
print(original_func.__doc__)     # 这是原始函数的文档字符串

上下文管理器

上下文管理器使用 with 语句，自动管理资源（如文件关闭）。

类实现

Python 
class FileManager:
    def __init__(self, filename, mode):
        self.filename = filename
        self.mode = mode
        self.file = None
    
    def __enter__(self):
        self.file = open(self.filename, self.mode)
        return self.file
    
    def __exit__(self, exc_type, exc_val, exc_tb):
        if self.file:
            self.file.close()
        return False  # 不处理异常

# 使用
with FileManager("test.txt", "w") as f:
    f.write("测试内容")
print("文件已关闭")

contextlib 实现

Python 
from contextlib import contextmanager

@contextmanager
def my_context():
    print("进入上下文")
    try:
        yield "资源"
    finally:
        print("退出上下文")

with my_context() as resource:
    print(f"使用 {resource}")

元类

元类用于控制类的创建，是 Python 中比较高级的概念。

Python 
class Meta(type):
    def __new__(cls, name, bases, attrs):
        # 在类创建时添加属性
        attrs['created_by'] = 'Meta'
        return super().__new__(cls, name, bases, attrs)

class MyClass(metaclass=Meta):
    pass

print(MyClass.created_by)  # Meta
