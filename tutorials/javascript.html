JavaScript 简介

JavaScript 是 Web 开发的核心语言，可用于前端（浏览器）和后端（Node.js）开发。

💡 JavaScript 特点
  • 动态类型语言
  • 基于原型的面向对象
  • 异步编程（Promise, async/await）
  • 函数式编程特性
  • 浏览器原生支持

变量和数据类型

变量声明

JavaScript 
var a = 10;          // 函数作用域
let b = 20;          // 块级作用域（推荐）
const PI = 3.14;     // 常量（不能修改）

// 基本类型
let str = "Hello";      // 字符串
let num = 123;          // 数字
let bool = true;        // 布尔
let undef = undefined;  // 未定义
let nul = null;         // 空值

console.log("字符串:", str);
console.log("数字:", num);
console.log("布尔:", bool);
字符串: Hello
数字: 123
布尔: true

模板字符串

JavaScript 
let name = "张三";
let age = 25;

// 模板字符串（反引号）
let greeting = `你好，我是${name}，今年${age}岁`;
console.log(greeting);
你好，我是张三，今年25岁

解构赋值

JavaScript 
// 数组解构
let [x, y] = [1, 2];
console.log(`x=${x}, y=${y}`);

// 对象解构
let { name, age } = { name: "李四", age: 30 };
console.log(`${name} - ${age}`);

// 展开运算符
let arr1 = [1, 2, 3];
let arr2 = [...arr1, 4, 5];
console.log(arr2);  // [1, 2, 3, 4, 5]

函数声明

JavaScript 
// 函数声明
function greet(name) {
    return `你好，${name}！`;
}

// 函数表达式

// 箭头函数
const multiply = (a, b) => a * b;

默认参数

JavaScript 
function power(base, exponent = 2) {
    return Math.pow(base, exponent);
}

console.log(power(3));    // 9
console.log(power(3, 3)); // 27

剩余参数

JavaScript 
function sum(...numbers) {
    return numbers.reduce((acc, cur) => acc + cur, 0);
}

console.log(sum(1, 2, 3));         // 6
console.log(sum(1, 2, 3, 4, 5));    // 15

数组方法

常用方法

JavaScript 
let nums = [1, 2, 3, 4, 5];

// map - 映射
let doubled = nums.map(n => n * 2);
console.log("map:", doubled);

// filter - 过滤
let evens = nums.filter(n => n % 2 === 0);
console.log("filter:", evens);

// reduce - 累计
let sum = nums.reduce((acc, n) => acc + n, 0);
console.log("reduce:", sum);

// find - 查找
let found = nums.find(n => n > 3);
console.log("find:", found);
map: [2, 4, 6, 8, 10]
filter: [2, 4]
reduce: 15
find: 4

面向对象

class 类

JavaScript 
class Person {
    constructor(name, age) {
        this.name = name;
        this.age = age;
    }

    greet() {
        console.log(`你好，我是${this.name}`);
    }
}

class Student extends Person {
    constructor(name, age, score) {
        super(name, age);
        this.score = score;
    }

    study() {
        console.log(`${this.name} 学习中，成绩: ${this.score}`);
    }
}

const s = new Student("小明", 18, 95);
s.greet();
s.study();
你好，我是小明
小明 学习中，成绩: 95

Promise 和 async/await

JavaScript 
// Promise
function delay(ms) {
    return new Promise(resolve => {
        setTimeout(() => resolve(`等待了${ms}ms`), ms);
    });
}

// async/await
async function main() {
    console.log("开始...");
    let result = await delay(100);
    console.log(result);
}

main();
