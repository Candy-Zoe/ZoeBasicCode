C 语言基础语法教程

C 语言简介

C 语言是一种通用、过程式的计算机编程语言，广泛用于系统软件开发，是很多编程语言的"母语"。

💡 C 语言特点
  • 静态类型语言，需要声明变量类型
  • 面向过程编程
  • 直接操作内存，适合系统级编程
  • 执行效率高，是很多语言的基石

第一个程序

C 
#include <stdio.h>

int main() {
    printf("Hello, World!\n");
    return 0;
}
⚠️ 编译运行 Windows: gcc hello.c -o hello && hello.exe
Linux/macOS: gcc hello.c -o hello && ./hello
Hello, World!

变量和数据类型

C 语言是静态类型语言，变量必须先声明类型才能使用。

基本数据类型

类型大小说明
char1字节字符型
int4字节整型（最常用）
float4字节单精度浮点型
double8字节双精度浮点型

变量声明

C 
int age = 25;           // 整型
float height = 1.75f;     // 单精度浮点型
double weight = 65.5;    // 双精度浮点型

printf("年龄: %d\n", age);
printf("身高: %.2f\n", height);
printf("体重: %.1lf\n", weight);
printf("等级: %c\n", grade);
年龄: 25
身高: 1.75
体重: 65.5
等级: A

算术运算符

自增自减运算符

C 
int i = 0;
printf("i = %d\n", i);
printf("i++ = %d\n", i++);  // 先返回再加
printf("i = %d\n", i);

++i = 2

位运算符

C 
int x = 6;  // 二进制: 0110
int y = 3;  // 二进制: 0011

printf("按位与: x & y = %d\n", x & y);  // 0010 = 2
printf("按位或: x | y = %d\n", x | y);  // 0111 = 7
printf("按位异或: x ^ y = %d\n", x ^ y); // 0101 = 5
printf("左移: x << 1 = %d\n", x << 1);  // 1100 = 12
printf("右移: x >> 1 = %d\n", x >> 1);  // 0011 = 3

循环语句

for 循环

C 
// 基本用法
for (int i = 0; i < 5; i++) {
    printf("第 %d 次循环\n", i + 1);
}

// 打印九九乘法表
printf("\n九九乘法表:\n");
for (int i = 1; i < 10; i++) {
    for (int j = 1; j <= i; j++) {

while 循环

C

do-while 循环

C 
int num = 1;
do {
    printf("num = %d\n", num);
    num++;
} while (num <= 5);
💡 do-while 特点 do-while 会先执行一次循环体，再判断条件，所以至少会执行一次。

break 和 continue

C 
// break: 跳出循环
for (int i = 0; i < 10; i++) {
    if (i == 5) break;
    printf("%d ", i);
}
printf("\n");

// continue: 跳过当前迭代
for (int i = 0; i < 10; i++) {
    if (i % 2 == 0) continue;
    printf("%d ", i);
}
0 1 2 3 4
1 3 5 7 9

递归函数

C 
int factorial(int n) {
    if (n == 0 || n == 1) {
        return 1;
    }
    return n * factorial(n - 1);
}

int main() {
    for (int i = 0; i < 6; i++) {
        printf("%d! = %d\n", i, factorial(i));
    }
    return 0;
}
0! = 1
1! = 1
2! = 2
3! = 6
4! = 24
5! = 120

结构体

定义和使用结构体

C 
#include <stdio.h>
#include <string.h>

struct Student {
    char name[50];
    int age;
    float score;
};

int main() {
    struct Student stu;
    strcpy(stu.name, "张三");
    stu.age = 20;
    stu.score = 95.5f;

    printf("学生信息:\n");
    printf("  姓名: %s\n", stu.name);
    printf("  年龄: %d\n", stu.age);
    printf("  成绩: %.1f\n", stu.score);

    return 0;
}
学生信息:
姓名: 张三
年龄: 20
成绩: 95.5

文件操作

读写文件

C 
#include <stdio.h>

int main() {
    // 写入文件
    FILE *fp = fopen("test.txt", "w");
    if (fp == NULL) {
        printf("打开文件失败\n");
        return 1;
    }
    fprintf(fp, "第一行\n");
    fprintf(fp, "第二行\n");
    fclose(fp);
    printf("写入成功\n");

    // 读取文件
    fp = fopen("test.txt", "r");
    char line[100];
    printf("文件内容:\n");
    while (fgets(line, sizeof(line), fp) != NULL) {
        printf("%s", line);
    }
    fclose(fp);

    return 0;
}
💡 常用文件模式
  • "r" - 读取
  • "w" - 写入（覆盖）
  • "a" - 追加
  • "rb"/"wb" - 二进制模式
