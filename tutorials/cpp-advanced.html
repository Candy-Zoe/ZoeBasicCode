类的继承

继承允许我们定义一个类继承另一个类的成员和方法。

基本继承

C++ 
#include 
#include 
using namespace std;

class Animal {
public:
    string name;

    Animal(const string& n) : name(n) {}

    void speak() {
        cout << name << " 发出声音" << endl;
    }
};

class Dog : public Animal {
public:
    Dog(const string& n) : Animal(n) {}

    void speak() {
        cout << name << " 汪汪叫" << endl;
    }
};

int main() {
    Dog dog("旺财");
    dog.speak();  // 调用 Dog::speak()
    return 0;
}
旺财 汪汪叫

protected 成员

C++ 
#include 
using namespace std;

class Base {
protected:
    int protected_data;  // 子类可访问，外部不可访问

public:
    Base() : protected_data(100) {}
};

class Derived : public Base {
public:
    void show() {
        cout << "子类访问: " << protected_data << endl;
    }
};

int main() {
    Derived d;
    d.show();
    // d.protected_data = 50;  // 错误！不可访问
    return 0;
}
子类访问: 100

多重继承

C++ 支持一个类继承多个父类。

菱形继承与虚继承

C++ 
#include 
using namespace std;

class Base {
public:
    int value;
};

class A : virtual public Base {};
class B : virtual public Base {};  // 虚继承，解决菱形继承问题

class C : public A, public B {};

int main() {
    C obj;
    obj.value = 42;
    cout << "value = " << obj.value << endl;
    return 0;
}
⚠️ 注意 菱形继承会导致基类数据重复，虚继承可以解决此问题。

虚函数

虚函数允许子类重写父类方法，实现多态。

基本虚函数

C++ 
#include 
using namespace std;

class Shape {
public:
    virtual void draw() {
        cout << "绘制图形" << endl;
    }

    virtual ~Shape() {}  // 虚析构函数
};

class Circle : public Shape {
public:
    void draw() override {
        cout << "绘制圆形" << endl;
    }
};

class Rectangle : public Shape {
public:
    void draw() override {
        cout << "绘制矩形" << endl;
    }
};

int main() {
    Shape *shapes[] = {new Circle(), new Rectangle()};

    for (int i = 0; i < 2; i++) {
        shapes[i]->draw();  // 多态调用
    }

    for (int i = 0; i < 2; i++) {
        delete shapes[i];
    }
    return 0;
}
绘制圆形
绘制矩形

虚表（vtable）原理

💡 原理说明 每个包含虚函数的类都有一个虚表（vtable），对象中包含指向虚表的指针（vptr）。调用虚函数时，通过 vptr 查找虚表来确定实际调用的函数。

抽象类

包含纯虚函数的类是抽象类，不能直接实例化。

C++ 
#include 
using namespace std;

class Shape {
public:
    virtual double area() const = 0;  // 纯虚函数
    virtual ~Shape() {}
};

class Circle : public Shape {
    double radius;
public:
    Circle(double r) : radius(r) {}

    double area() const override {
        return 3.14159 * radius * radius;
    }
};

class Rectangle : public Shape {
    double width, height;
public:
    Rectangle(double w, double h) : width(w), height(h) {}

    double area() const override {
        return width * height;
    }
};

int main() {
    // Shape s;  // 错误！抽象类不能实例化

    Shape* shapes[] = {new Circle(5), new Rectangle(4, 5)};

    cout << "圆形面积: " << shapes[0]->area() << endl;
    cout << "矩形面积: " << shapes[1]->area() << endl;

    delete shapes[0];
    delete shapes[1];
    return 0;
}
圆形面积: 78.5397
矩形面积: 20

运算符重载

C++ 
#include 
using namespace std;

class Point {
public:
    int x, y;

    Point(int _x = 0, int _y = 0) : x(_x), y(_y) {}

    // 运算符重载
    Point operator+(const Point& other) const {
        return Point(x + other.x, y + other.y);
    }

    bool operator==(const Point& other) const {
        return x == other.x && y == other.y;
    }
};

int main() {
    Point p1(1, 2);
    Point p2(3, 4);
    Point p3 = p1 + p2;

    cout << "p3 = (" << p3.x << ", " << p3.y << ")" << endl;
    cout << "p1 == p2: " << (p1 == p2) << endl;
    return 0;
}
p3 = (4, 6)
p1 == p2: 0

类模板

C++ 
#include 
using namespace std;

template 
class Stack {
private:
    vector elements;
public:
    void push(const T& elem) {
        elements.push_back(elem);
    }

    T pop() {
        T top = elements.back();
        elements.pop_back();
        return top;
    }

    bool empty() const {
        return elements.empty();
    }
};

int main() {
    Stack int_stack;
    int_stack.push(10);
    int_stack.push(20);
    cout << int_stack.pop() << endl;  // 20

    Stack str_stack;
    str_stack.push("Hello");
    str_stack.push("World");
    cout << str_stack.pop() << endl;  // World
    return 0;
}
20
World

vector（动态数组）

C++ 
#include 
#include 
using namespace std;

int main() {
    vector vec = {1, 2, 3};

    vec.push_back(4);
    vec.push_back(5);

    cout << "大小: " << vec.size() << endl;
    cout << "元素: ";
    for (int i = 0; i < vec.size(); i++) {
        cout << vec[i] << " ";
    }
    cout << endl;

    // 迭代器遍历
    cout << "迭代器: ";
    for (auto it = vec.begin(); it != vec.end(); it++) {
        cout << *it << " ";
    }
    cout << endl;
    return 0;
}

STL 算法

C++ 
#include 
#include 
#include 
using namespace std;

int main() {
    vector v = {5, 2, 8, 1, 9, 3};

    // 排序
    sort(v.begin(), v.end());
    cout << "排序后: ";
    for (int x : v) cout << x << " ";
    cout << endl;

    // 查找
    auto it = find(v.begin(), v.end(), 8);
    if (it != v.end()) {
        cout << "找到 8 在位置 " << (it - v.begin()) << endl;
    }

    // 统计满足条件的元素
    int count = count_if(v.begin(), v.end(), [](int x) { return x > 5; });
    cout << "大于5的元素个数: " << count << endl;

    // transform
    vector squares(v.size());
    transform(v.begin(), v.end(), squares.begin(), [](int x) { return x * x; });
    cout << "平方: ";
    for (int x : squares) cout << x << " ";
    cout << endl;
    return 0;
}
排序后: 1 2 3 5 8 9
找到 8 在位置 4
大于5的元素个数: 2
平方: 1 4 9 25 64 81

智能指针

智能指针自动管理内存，避免内存泄漏。

unique_ptr

C++ 
#include 
#include

unique_ptr create_value() {
    return make_unique(42);
}

int main() {
    // 创建
    unique_ptr p1 = make_unique(100);

    // 使用
    cout << "值: " << *p1 << endl;

    unique_ptr p2 = move(p1);
    // cout << *p1 << endl;  // 错误！p1 已为空
    cout << "新值: " << *p2 << endl;

异常处理

C++ 
#include 
#include 
using namespace std;

class MyException : public exception {
public:
    const char* what() const noexcept override {
        return "自定义异常";
    }
};

void divide(int a, int b) {
    if (b == 0) {
        throw runtime_error("除数不能为零");
    }
    cout << "结果: " << a / b << endl;
}

int main() {
    try {
        divide(10, 2);
        divide(10, 0);
    } catch (const runtime_error& e) {
        cout << "捕获异常: " << e.what() << endl;
    } catch (...) {
        cout << "捕获未知异常" << endl;
    }
    return 0;
}
结果: 5
捕获异常: 除数不能为零

Lambda 表达式

Lambda 是匿名函数，可以捕获外部变量。

C++ 
#include 
#include 
#include 
using namespace std;

int main() {
    int x = 10;
    int y = 20;

    // 值捕获 [x]
    // 引用捕获 [&y]
    // 全部捕获 [=] [&]

    auto add = [x, y]() {
        return x + y;
    };

    cout << "和: " << add() << endl;

    // 可变 lambda（修改捕获的变量）
    auto increment = [x]() mutable {
        x += 10;
        return x;
    };

    cout << "增量: " << increment() << endl;
    cout << "原值: " << x << endl;  // x 未改变

    // 带参数的 lambda
    vector v = {3, 1, 4, 1, 5, 9, 2};
    sort(v.begin(), v.end(), [](int a, int b) {
        return a > b;  // 降序
    });
    cout << "降序: ";
    for (int n : v) cout << n << " ";
    cout << endl;
    return 0;
}
和: 30
增量: 20
原值: 10
降序: 9 5 4 3 2 1 1
