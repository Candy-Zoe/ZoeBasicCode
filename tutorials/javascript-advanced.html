原型链

JavaScript 通过原型链实现继承，每个对象都有一个 prototype 属性。

原型基础

JavaScript 
// 构造函数
function Person(name, age) {
    this.name = name;
    this.age = age;
}

// 在原型上添加方法
Person.prototype.greet = function() {
    console.log(`你好，我是${this.name}`);
};

const p1 = new Person("张三", 25);
const p2 = new Person("李四", 30);

p1.greet();  // 通过原型调用
p2.greet();

console.log(p1.__proto__ === Person.prototype);  // true
console.log(Person.prototype.constructor === Person);  // true
你好，我是张三
你好，我是李四
true

原型链

JavaScript 
function Parent() {
    this.parentName = "Parent";
}

function Child() {
    this.childName = "Child";
}

// 设置原型链
Child.prototype = new Parent();
Child.prototype.constructor = Child;

const child = new Child();
console.log(child.parentName);  // Parent
console.log(child.childName);   // Child
console.log(child instanceof Child);   // true
console.log(child instanceof Parent);  // true

class 类 (ES6)

ES6 引入的 class 语法是构造函数的语法糖。

基本用法

JavaScript 
class Person {
    // 构造函数
    constructor(name, age) {
        this.name = name;
        this.age = age;
    }

    // 方法
    greet() {
        console.log(`你好，我是${this.name}`);
    }

    // 静态方法
    static create(name, age) {
        return new Person(name, age);
    }

    // getter
    get info() {
        return `${this.name}, ${this.age}岁`;
    }
}

const p = new Person("张三", 25);
p.greet();
console.log(p.info);
console.log(Person.create("李四", 30));
你好，我是张三
张三, 25岁

私有字段

JavaScript 
class BankAccount {
    // 私有字段（ES2022）
    #balance = 0;

    deposit(amount) {
        this.#balance += amount;
    }

    withdraw(amount) {
        if (amount <= this.#balance) {
            this.#balance -= amount;
        }
    }

    getBalance() {
        return this.#balance;
    }
}

const account = new BankAccount();
account.deposit(1000);
account.withdraw(300);
console.log(account.getBalance());  // 700
// console.log(account.#balance);  // 语法错误

继承

extends 继承

JavaScript 
class Animal {
    constructor(name) {
        this.name = name;
    }

    speak() {
        console.log(`${this.name} 发出声音`);
    }
}

class Dog extends Animal {
    constructor(name, breed) {
        super(name);  // 调用父类构造函数
        this.breed = breed;
    }

    speak() {
        console.log(`${this.name} 汪汪叫`);
    }

    fetch() {
        console.log(`${this.name} 去捡球`);
    }
}

const dog = new Dog("旺财", "柴犬");
dog.speak();  // 重写的方法
dog.fetch();  // 子类方法
super.speak?.();  // 调用父类方法（需在方法内）
旺财 汪汪叫
旺财 去捡球

多层继承

JavaScript 
class A {
    method() { console.log("A"); }
}

class B extends A {
    method() { console.log("B"); }
}

class C extends B {
    method() { console.log("C"); }
}

const c = new C();
c.method();  // C
console.log(c instanceof A);  // true
console.log(c instanceof B);  // true
console.log(c instanceof C);  // true

常见应用

JavaScript 
// 工厂函数
function multiply(factor) {
    return function(num) {
        return num * factor;
    };
}

const double = multiply(2);
const triple = multiply(3);

console.log(double(5));   // 10
console.log(triple(5));    // 15

// 私有方法
function Person(name) {
    let _age = 0;  // 私有变量
    
    this.getAge = function() {
        return _age;
    };
    
    this.setAge = function(age) {
        if (age >= 0) _age = age;
    };
}

const p = new Person("张三");
p.setAge(25);
console.log(p.getAge());  // 25

Promise

Promise 用于处理异步操作，是回调函数的替代方案。

创建 Promise

JavaScript 
const promise = new Promise((resolve, reject) => {
    const success = true;
    
    if (success) {
        resolve("操作成功");
    } else {
        reject("操作失败");
    }
});

promise
    .then(result => console.log(result))
    .catch(error => console.error(error))
    .finally(() => console.log("完成"));
操作成功
完成

链式调用

JavaScript 
fetch("/api/user")
    .then(response => response.json())
    .then(user => {
        console.log(user);
        return fetch(`/api/posts/${user.id}`);
    })
    .then(response => response.json())
    .then(posts => console.log(posts))
    .catch(error => console.error(error));

并行处理

JavaScript 
const p1 = Promise.resolve(1);
const p2 = Promise.resolve(2);
const p3 = Promise.resolve(3);

// 等待所有完成
Promise.all([p1, p2, p3])
    .then(results => console.log(results));  // [1, 2, 3]

// 任意一个完成（最先完成的）
Promise.race([p1, p2, p3])
    .then(result => console.log(result));  // 1

// 任意一个完成（第一个成功的）
Promise.any([p1, p2, p3])
    .then(result => console.log(result));  // 1

// 所有完成（无论成功失败）
Promise.allSettled([p1, p2, p3])
    .then(results => console.log(results));

async/await

async/await 是 Promise 的语法糖，让异步代码看起来像同步代码。

基本用法

JavaScript 
// async 函数总是返回 Promise
async function fetchUser(id) {
    const response = await fetch(`/api/users/${id}`);
    const user = await response.json();
    return user;
}

async function main() {
    try {
        const user = await fetchUser(1);
        console.log(user);
    } catch (error) {
        console.error("获取用户失败:", error);
    }
}

main();

并发执行

JavaScript 
async function getUsers(ids) {
    // 并发请求
    const promises = ids.map(id => fetchUser(id));
    const users = await Promise.all(promises);
    return users;
}

// 顺序执行（错误示例）
async function wrongOrder() {
    const user1 = await fetchUser(1);
    const user2 = await fetchUser(2);  // 等待 user1 完成
}

// 并发执行（正确示例）
async function rightOrder() {
    const [user1, user2] = await Promise.all([
        fetchUser(1),
        fetchUser(2)
    ]);
}

Proxy 代理

Proxy 用于定义对象的自定义行为。

JavaScript 
const target = {
    name: "张三",
    age: 25
};

const handler = {
    get(target, prop) {
        console.log(`获取属性: ${prop}`);
        return target[prop];
    },
    set(target, prop, value) {
        console.log(`设置属性: ${prop} = ${value}`);
        target[prop] = value;
        return true;
    }
};

const proxy = new Proxy(target, handler);

console.log(proxy.name);  // 获取属性: name\n张三
proxy.age = 30;          // 设置属性: age = 30

响应式系统

JavaScript 
function reactive(obj) {
    return new Proxy(obj, {
        get(target, prop) {
            return Reflect.get(target, prop);
        },
        set(target, prop, value) {
            console.log(`属性 ${prop} 从 ${target[prop]} 变为 ${value}`);
            return Reflect.set(target, prop, value);
        }
    });
}

const state = reactive({ count: 0 });
state.count++;  // 属性 count 从 0 变为 1

Symbol

Symbol 是一种唯一标识符，常用于对象属性名。

JavaScript 
// 创建 Symbol
const sym1 = Symbol("description");
const sym2 = Symbol("description");
console.log(sym1 === sym2);  // false

// 作为对象属性
const obj = {
    [sym1]: "值1",
    [sym2]: "值2"
};

// 常用 Symbol
Symbol.iterator  // 可迭代对象
Symbol.toStringTag  // toString 标签
Symbol.hasInstance  // instanceof

// 全局 Symbol
const globalSym = Symbol.for("global");
const sameSym = Symbol.for("global");
console.log(globalSym === sameSym);  // true

生成器

生成器函数使用 yield 产生一系列值。

JavaScript 
function* numberGenerator() {
    yield 1;
    yield 2;
    yield 3;
}

const gen = numberGenerator();
console.log(gen.next());  // { value: 1, done: false }
console.log(gen.next());  // { value: 2, done: false }
console.log(gen.next());  // { value: 3, done: false }
console.log(gen.next());  // { value: undefined, done: true }

// 可迭代
for (const num of numberGenerator()) {
    console.log(num);
}

{ value: 3, done: false }
