SQL 基础教程

SQL 简介

SQL（Structured Query Language，结构化查询语言）是用于管理关系数据库的标准语言。所有主流数据库（MySQL、PostgreSQL、SQLite、Oracle、SQL Server）都支持 SQL。

💡 SQL 分类
  • DDL：数据定义语言（CREATE、ALTER、DROP）
  • DML：数据操作语言（INSERT、UPDATE、DELETE）
  • DQL：数据查询语言（SELECT）
  • DCL：数据控制语言（GRANT、REVOKE）
  • TCL：事务控制语言（COMMIT、ROLLBACK）

DDL 数据定义语言

创建和删除数据库

SQL
-- 创建数据库
CREATE DATABASE mydb DEFAULT CHARACTER SET utf8mb4;

-- 删除数据库
DROP DATABASE IF EXISTS mydb;

-- 使用数据库
USE mydb;

创建表

SQL
CREATE TABLE students (
    id INT PRIMARY KEY AUTO_INCREMENT,
    name VARCHAR(50) NOT NULL,
    age INT,
    email VARCHAR(100) UNIQUE,
    grade VARCHAR(20) DEFAULT '一年级',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE courses (
    id INT PRIMARY KEY AUTO_INCREMENT,
    course_name VARCHAR(100) NOT NULL,
    teacher VARCHAR(50),
    credits INT DEFAULT 3
);

CREATE TABLE scores (
    id INT PRIMARY KEY AUTO_INCREMENT,
    student_id INT,
    course_id INT,
    score DECIMAL(5,2),
    FOREIGN KEY (student_id) REFERENCES students(id),
    FOREIGN KEY (course_id) REFERENCES courses(id)
);
⚠️ 常用数据类型
  • 整数：INT、BIGINT、TINYINT、SMALLINT
  • 小数：DECIMAL(总位数, 小数位)、FLOAT、DOUBLE
  • 字符串：VARCHAR(长度)、TEXT、CHAR(长度)
  • 日期：DATE、TIME、DATETIME、TIMESTAMP
  • 布尔：BOOLEAN（实际是 TINYINT(1)）

修改表结构

SQL
-- 添加列
ALTER TABLE students ADD COLUMN phone VARCHAR(20);

-- 修改列
ALTER TABLE students MODIFY COLUMN name VARCHAR(80);

-- 删除列
ALTER TABLE students DROP COLUMN phone;

-- 查看表结构
DESCRIBE students;

-- 删除表
DROP TABLE IF EXISTS scores;

DML 数据操作语言

插入数据

SQL
-- 插入单行
INSERT INTO students (name, age, email, grade)
VALUES ('张三', 18, 'zhangsan@example.com', '高三');

-- 插入多行
INSERT INTO students (name, age, email, grade) VALUES
    ('李四', 17, 'lisi@example.com', '高二'),
    ('王五', 16, 'wangwu@example.com', '高一'),
    ('赵六', 18, 'zhaoliu@example.com', '高三');

更新数据

SQL
-- 更新单行
UPDATE students SET grade = '高三' WHERE id = 5;

-- 批量更新
UPDATE scores SET score = score + 5 WHERE score < 60;

-- 多列更新
UPDATE students SET age = age + 1, grade = '新年级' WHERE id = 1;

删除数据

SQL
-- 删除指定行
DELETE FROM students WHERE id = 1;

-- 删除所有行（逐行删除，可回滚）
DELETE FROM students;

-- 清空表（直接删除整表，不可回滚，更快）
TRUNCATE TABLE students;
🚨 注意事项
  • UPDATE 和 DELETE 一定要带 WHERE 条件！
  • 执行前先 SELECT 确认范围
  • 重要操作开启事务，确认无误再提交

查询语句 SELECT

基础查询

SQL
-- 查询所有列
SELECT * FROM students;

-- 查询指定列
SELECT name, age, email FROM students;

-- 别名
SELECT name AS 姓名, age AS 年龄 FROM students;

-- 去重
SELECT DISTINCT grade FROM students;

-- 限制行数
SELECT * FROM students LIMIT 10;       -- 前10条
SELECT * FROM students LIMIT 10, 5;    -- 跳过10条，取5条

WHERE 条件

SQL
-- 比较运算符
SELECT * FROM students WHERE age > 17;
SELECT * FROM students WHERE age <= 18;

-- 逻辑运算符
SELECT * FROM students WHERE grade = '高三' AND age >= 18;
SELECT * FROM students WHERE age = 17 OR age = 18;
SELECT * FROM students WHERE NOT active = 0;

-- 范围查询
SELECT * FROM students WHERE age BETWEEN 16 AND 18;

-- IN 查询
SELECT * FROM students WHERE grade IN ('高三', '高二');

-- 模糊查询
SELECT * FROM students WHERE name LIKE '张%';   -- 张开头
SELECT * FROM students WHERE name LIKE '%三%';  -- 包含三

-- 空值判断
SELECT * FROM students WHERE email IS NULL;
SELECT * FROM students WHERE email IS NOT NULL;

排序 ORDER BY

SQL
-- 升序（默认）
SELECT * FROM students ORDER BY age ASC;

-- 降序
SELECT * FROM students ORDER BY age DESC;

-- 多字段排序
SELECT * FROM students ORDER BY grade ASC, age DESC;

聚合函数与分组

SQL
-- 聚合函数
SELECT COUNT(*) AS 总人数 FROM students;
SELECT AVG(age) AS 平均年龄 FROM students;
SELECT MAX(score) AS 最高分 FROM scores;
SELECT MIN(score) AS 最低分 FROM scores;
SELECT SUM(score) AS 总分 FROM scores WHERE student_id = 1;

-- GROUP BY 分组
SELECT grade, COUNT(*) AS 人数
FROM students
GROUP BY grade;

-- HAVING 分组后过滤
SELECT student_id, AVG(score) AS 平均分
FROM scores
GROUP BY student_id
HAVING AVG(score) > 90;

连接查询 JOIN

SQL
-- INNER JOIN 内连接（两边都匹配）
SELECT s.name, c.course_name, sc.score
FROM scores sc
INNER JOIN students s ON sc.student_id = s.id
INNER JOIN courses c ON sc.course_id = c.id;

-- LEFT JOIN 左连接（左表全部，右表匹配）
SELECT s.name, AVG(sc.score) AS 平均分
FROM students s
LEFT JOIN scores sc ON s.id = sc.student_id
GROUP BY s.id;

-- RIGHT JOIN 右连接
SELECT c.course_name, COUNT(sc.id) AS 选课人数
FROM scores sc
RIGHT JOIN courses c ON sc.course_id = c.id
GROUP BY c.id;

-- 自连接
SELECT a.name, b.name
FROM students a, students b
WHERE a.grade = b.grade AND a.id <> b.id;
连接类型说明
INNER JOIN只返回两边都匹配的行
LEFT JOIN返回左表所有行，右表匹配的行
RIGHT JOIN返回右表所有行，左表匹配的行
FULL JOIN返回两边所有行（部分数据库支持）
CROSS JOIN笛卡尔积，所有组合

子查询

SQL
-- 标量子查询
SELECT name, (
    SELECT AVG(score) FROM scores WHERE student_id = s.id
) AS 平均分
FROM students s;

-- IN 子查询
SELECT * FROM students
WHERE id IN (
    SELECT DISTINCT student_id FROM scores WHERE score > 95
);

-- EXISTS 子查询
SELECT * FROM students s
WHERE EXISTS (
    SELECT 1 FROM scores sc
    WHERE sc.student_id = s.id AND sc.score > 95
);

-- FROM 子查询
SELECT name, 平均分
FROM (
    SELECT s.name, AVG(sc.score) AS 平均分
    FROM students s
    JOIN scores sc ON s.id = sc.student_id
    GROUP BY s.id
) AS student_avg
WHERE 平均分 > 90;

高级特性

CASE WHEN

SQL
SELECT
    name,
    CASE
        WHEN age < 16 THEN '少年'
        WHEN age < 18 THEN '青年'
        ELSE '成年'
    END AS 年龄段
FROM students;

常用函数

SQL
-- 字符串函数
SELECT UPPER(name), LOWER(name), LENGTH(name) FROM students;
SELECT CONCAT(name, ' - ', grade) FROM students;
SELECT SUBSTRING(name, 1, 1) FROM students;

-- 数学函数
SELECT ROUND(95.678, 2), CEIL(95.1), FLOOR(95.9), ABS(-10);

-- 日期函数
SELECT NOW(), CURDATE(), CURTIME();
SELECT YEAR(NOW()), MONTH(NOW()), DAY(NOW());
SELECT DATE_ADD(NOW(), INTERVAL 1 DAY);

窗口函数 (MySQL 8.0+)

SQL
-- 排名
SELECT
    name,
    grade,
    ROW_NUMBER() OVER (PARTITION BY grade ORDER BY age DESC) AS rn
FROM students;

-- 其他窗口函数：RANK(), DENSE_RANK()
-- SUM() OVER(), AVG() OVER() 等聚合窗口

索引与事务

索引

SQL
-- 创建索引
CREATE INDEX idx_student_name ON students(name);
CREATE INDEX idx_score_student ON scores(student_id);

-- 唯一索引
CREATE UNIQUE INDEX idx_student_email ON students(email);

-- 组合索引（最左前缀原则）
CREATE INDEX idx_name_age ON students(name, age);

-- 查看索引
SHOW INDEX FROM students;

-- 删除索引
DROP INDEX idx_student_name ON students;

-- 查看执行计划
EXPLAIN SELECT * FROM students WHERE name = '张三';

事务

SQL
-- 开始事务
START TRANSACTION;
BEGIN;

-- 执行 SQL
UPDATE accounts SET balance = balance - 100 WHERE id = 1;
UPDATE accounts SET balance = balance + 100 WHERE id = 2;

-- 设置保存点
SAVEPOINT sp1;

-- 回滚到保存点
ROLLBACK TO sp1;

-- 提交事务
COMMIT;

-- 回滚事务
ROLLBACK;
💡 事务四大特性 (ACID)
  • 原子性 (Atomicity)：要么全成功，要么全失败
  • 一致性 (Consistency)：事务前后数据保持一致
  • 隔离性 (Isolation)：并发事务互不干扰
  • 持久性 (Durability)：提交后永久保存
