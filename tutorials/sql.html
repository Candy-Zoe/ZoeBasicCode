SQL 基础教程

SQL 简介

SQL（Structured Query Language，结构化查询语言）是用于管理关系数据库的标准语言。所有主流数据库（MySQL、PostgreSQL、SQLite、Oracle、SQL Server）都支持 SQL。

💡 SQL 分类
  • DDL：数据定义语言（CREATE、ALTER、DROP）
  • DML：数据操作语言（INSERT、UPDATE、DELETE）
  • DQL：数据查询语言（SELECT）
  • DCL：数据控制语言（GRANT、REVOKE）
  • TCL：事务控制语言（COMMIT、ROLLBACK）

DDL 数据定义语言

创建和删除数据库

SQL
-- 创建数据库
CREATE DATABASE mydb DEFAULT CHARACTER SET utf8mb4;

-- 删除数据库
DROP DATABASE IF EXISTS mydb;

-- 使用数据库
USE mydb;

创建表

SQL
CREATE TABLE students (
    id INT PRIMARY KEY AUTO_INCREMENT,
    name VARCHAR(50) NOT NULL,
    age INT,
    email VARCHAR(100) UNIQUE,
    grade VARCHAR(20) DEFAULT '一年级',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE courses (
    id INT PRIMARY KEY AUTO_INCREMENT,
    course_name VARCHAR(100) NOT NULL,
    teacher VARCHAR(50),
    credits INT DEFAULT 3
);

CREATE TABLE scores (
    id INT PRIMARY KEY AUTO_INCREMENT,
    student_id INT,
    course_id INT,
    score DECIMAL(5,2),
    FOREIGN KEY (student_id) REFERENCES students(id),
    FOREIGN KEY (course_id) REFERENCES courses(id)
);
⚠️ 常用数据类型
  • 整数：INT、BIGINT、TINYINT、SMALLINT
  • 小数：DECIMAL(总位数, 小数位)、FLOAT、DOUBLE
  • 字符串：VARCHAR(长度)、TEXT、CHAR(长度)
  • 日期：DATE、TIME、DATETIME、TIMESTAMP
  • 布尔：BOOLEAN（实际是 TINYINT(1)）

修改表结构

SQL
-- 添加列
ALTER TABLE students ADD COLUMN phone VARCHAR(20);

-- 修改列
ALTER TABLE students MODIFY COLUMN name VARCHAR(80);

-- 删除列
ALTER TABLE students DROP COLUMN phone;

-- 查看表结构
DESCRIBE students;

-- 删除表
DROP TABLE IF EXISTS scores;

DML 数据操作语言

插入数据

SQL
-- 插入单行
INSERT INTO students (name, age, email, grade)
VALUES ('张三', 18, 'zhangsan@example.com', '高三');

-- 插入多行
INSERT INTO students (name, age, email, grade) VALUES
    ('李四', 17, 'lisi@example.com', '高二'),
    ('王五', 16, 'wangwu@example.com', '高一'),
    ('赵六', 18, 'zhaoliu@example.com', '高三');

更新数据

SQL
-- 更新单行
UPDATE students SET grade = '高三' WHERE id = 5;

-- 批量更新
UPDATE scores SET score = score + 5 WHERE score < 60;

-- 多列更新
UPDATE students SET age = age + 1, grade = '新年级' WHERE id = 1;

删除数据

SQL
-- 删除指定行
DELETE FROM students WHERE id = 1;

-- 删除所有行（逐行删除，可回滚）
DELETE FROM students;

-- 清空表（直接删除整表，不可回滚，更快）
TRUNCATE TABLE students;
🚨 注意事项
  • UPDATE 和 DELETE 一定要带 WHERE 条件！
  • 执行前先 SELECT 确认范围
  • 重要操作开启事务，确认无误再提交

查询语句 SELECT

SELECT 是 SQL 中最强大、最常用的语句，用于从数据库中查询数据。完整的 SELECT 语句执行顺序如下：

💡 SELECT 语句执行顺序
  1. FROM - 确定数据来源表
  2. WHERE - 筛选满足条件的行
  3. GROUP BY - 对结果分组
  4. HAVING - 筛选分组后的结果
  5. SELECT - 选择要显示的列
  6. ORDER BY - 对结果排序
  7. LIMIT - 限制返回行数

基础查询

SQL
-- 查询所有列（* 不推荐在生产环境使用）
SELECT * FROM students;

-- 查询指定列（推荐，明确知道要什么）
SELECT name, age, email FROM students;

-- 别名：让列名更易读
SELECT name AS 姓名, age AS 年龄, email AS 邮箱 FROM students;

-- 去重：消除重复行
SELECT DISTINCT grade FROM students;

-- 限制行数（不同数据库语法略有不同）
SELECT * FROM students LIMIT 10;       -- MySQL: 前10条
SELECT * FROM students LIMIT 10, 5;     -- MySQL: 跳过10条，取5条
SELECT * FROM students FETCH FIRST 10 ROWS ONLY;  -- Oracle/SQL Server

WHERE 条件查询

WHERE 子句用于筛选满足条件的记录，是 SELECT 中最常用的过滤方式。

SQL
-- 比较运算符：= > < >= <= <> !=
SELECT * FROM students WHERE age > 17;      -- 大于
SELECT * FROM students WHERE age <= 18;     -- 小于等于
SELECT * FROM students WHERE grade <> '高三'; -- 不等于

-- 逻辑运算符：AND、OR、NOT
SELECT * FROM students WHERE grade = '高三' AND age >= 18;  -- 两个条件都满足
SELECT * FROM students WHERE age = 17 OR age = 18;         -- 满足任一条件
SELECT * FROM students WHERE NOT grade = '高三';              -- 取反

-- 范围查询：BETWEEN...AND
SELECT * FROM students WHERE age BETWEEN 16 AND 18;  -- 包含两端

-- IN 查询：在列表中
SELECT * FROM students WHERE grade IN ('高三', '高二', '高一');

-- 模糊查询：LIKE（% 匹配任意字符，_ 匹配单个字符）
SELECT * FROM students WHERE name LIKE '张%';    -- 张开头（张三、张三丰）
SELECT * FROM students WHERE name LIKE '%三%';   -- 包含三
SELECT * FROM students WHERE name LIKE '张_';    -- 张后面一个字（张飞、张三）
SELECT * FROM students WHERE name LIKE '%三';    -- 以三结尾

-- 空值判断：IS NULL / IS NOT NULL
SELECT * FROM students WHERE email IS NULL;      -- 邮箱为空
SELECT * FROM students WHERE email IS NOT NULL;  -- 邮箱不为空

-- 组合示例：综合查询
SELECT name, grade, age
FROM students
WHERE grade IN ('高三', '高二')
  AND age >= 17
  AND name LIKE '张%'
ORDER BY grade ASC, age DESC
LIMIT 20;

ORDER BY 排序

ORDER BY 用于对查询结果进行排序，ASC 升序（默认），DESC 降序。

SQL
-- 升序排列（可以省略 ASC，默认就是升序）
SELECT * FROM students ORDER BY age ASC;
SELECT * FROM students ORDER BY age;  -- 效果相同

-- 降序排列
SELECT * FROM students ORDER BY age DESC;

-- 多字段排序：先按第一个字段排，相同时按第二个字段排
SELECT * FROM students ORDER BY grade ASC, age DESC;

-- 按表达式排序
SELECT name, score, score * 1.1 AS 加分后 FROM scores ORDER BY 加分后 DESC;

-- 按列位置排序（不推荐，可读性差）
SELECT name, grade, age FROM students ORDER BY 2 ASC, 3 DESC;

LIMIT 分页查询

LIMIT 用于限制返回的行数，常用于分页查询。

SQL
-- 前 N 条记录
SELECT * FROM students LIMIT 10;

-- 分页查询（每页 10 条，获取第 1 页）
SELECT * FROM students ORDER BY id LIMIT 10 OFFSET 0;

-- 分页查询（每页 10 条，获取第 2 页）
SELECT * FROM students ORDER BY id LIMIT 10 OFFSET 10;

-- 分页查询（每页 10 条，获取第 3 页）
SELECT * FROM students ORDER BY id LIMIT 10 OFFSET 20;

-- 获取成绩最高的前 5 名
SELECT student_id, score
FROM scores
ORDER BY score DESC
LIMIT 5;
💡 分页查询优化建议
  • 分页查询一定要有 ORDER BY，确保结果稳定
  • 大偏移量时性能下降，可用子查询优化
  • 记录总数可以用 SQL_CALC_FOUND_ROWS（MySQL）或窗口函数

聚合函数与 GROUP BY 分组

聚合函数用于对一组值进行计算，返回单个值。常与 GROUP BY 一起使用。

SQL
-- 常用聚合函数
SELECT COUNT(*) AS 总人数 FROM students;           -- 计数
SELECT AVG(age) AS 平均年龄 FROM students;          -- 平均值
SELECT MAX(score) AS 最高分 FROM scores;           -- 最大值
SELECT MIN(score) AS 最低分 FROM scores;           -- 最小值
SELECT SUM(score) AS 总分 FROM scores WHERE student_id = 1;  -- 求和
SELECT ROUND(AVG(score), 2) AS 平均分 FROM scores; -- 四舍五入

-- GROUP BY 分组：按年级统计人数
SELECT grade, COUNT(*) AS 人数
FROM students
GROUP BY grade;

-- 多字段分组
SELECT grade, gender, COUNT(*) AS 人数
FROM students
GROUP BY grade, gender;

-- GROUP_CONCAT：合并分组内的值
SELECT grade, GROUP_CONCAT(name) AS 班级同学
FROM students
GROUP BY grade;

HAVING 分组后过滤

HAVING 与 WHERE 的区别：WHERE 在分组前过滤，HAVING 在分组后过滤。

SQL
-- HAVING：筛选平均分大于 90 的学生
SELECT student_id, AVG(score) AS 平均分
FROM scores
GROUP BY student_id
HAVING AVG(score) > 90;

-- WHERE vs HAVING 的区别
-- WHERE：在分组前过滤（先筛选，后分组）
SELECT grade, COUNT(*) AS 人数
FROM students
WHERE age > 17
GROUP BY grade;

-- HAVING：在分组后过滤（先分组，后筛选）
SELECT grade, COUNT(*) AS 人数
FROM students
GROUP BY grade
HAVING COUNT(*) > 5;

-- 综合示例：统计各年级平均分，筛选平均分大于 85 的年级
SELECT
    s.grade AS 年级,
    COUNT(*) AS 人数,
    ROUND(AVG(sc.score), 2) AS 平均分
FROM students s
INNER JOIN scores sc ON s.id = sc.student_id
GROUP BY s.grade
HAVING AVG(sc.score) > 85
ORDER BY 平均分 DESC;
⚠️ 重要区别
  • WHERE：过滤原始数据，在 GROUP BY 之前执行
  • HAVING：过滤分组后的聚合数据，在 GROUP BY 之后执行
  • 聚合函数（SUM、AVG、COUNT 等）不能用在 WHERE 中，需要用 HAVING

连接查询 JOIN

JOIN 用于将多个表中的数据根据条件组合在一起，是关系型数据库的核心操作。

SQL
-- INNER JOIN 内连接（两边都匹配才返回）
-- 查询学生姓名、课程名称、成绩
SELECT s.name AS 学生姓名, c.course_name AS 课程名称, sc.score AS 成绩
FROM scores sc
INNER JOIN students s ON sc.student_id = s.id
INNER JOIN courses c ON sc.course_id = c.id;

-- LEFT JOIN 左连接（以左表为主，右表没有匹配则为 NULL）
-- 查询所有学生及其平均分（包括没有成绩的学生）
SELECT s.name AS 学生姓名,
       COALESCE(AVG(sc.score), 0) AS 平均分  -- COALESCE 处理 NULL 值
FROM students s
LEFT JOIN scores sc ON s.id = sc.student_id
GROUP BY s.id, s.name;

-- RIGHT JOIN 右连接（以右表为主，左表没有匹配则为 NULL）
-- 查询所有课程及选课人数（包括没人选的课）
SELECT c.course_name AS 课程名称, COUNT(sc.id) AS 选课人数
FROM scores sc
RIGHT JOIN courses c ON sc.course_id = c.id
GROUP BY c.id, c.course_name;

-- FULL OUTER JOIN 全连接（两边所有记录）
-- MySQL 不直接支持，可用 UNION 模拟
SELECT s.name, c.course_name
FROM students s
LEFT JOIN scores sc ON s.id = sc.student_id
LEFT JOIN courses c ON sc.course_id = c.id
UNION
SELECT s.name, c.course_name
FROM students s
RIGHT JOIN scores sc ON s.id = sc.student_id
RIGHT JOIN courses c ON sc.course_id = c.id;

自连接

自连接是表与自身的连接，常用于处理层次结构数据（如员工表、分类表）。

SQL
-- 假设有员工表 employees，包含员工ID、姓名、上级ID
CREATE TABLE employees (
    id INT PRIMARY KEY,
    name VARCHAR(50),
    manager_id INT,
    FOREIGN KEY (manager_id) REFERENCES employees(id)
);

-- 查询每个员工及其上级姓名
SELECT e.name AS 员工, m.name AS 上级
FROM employees e
LEFT JOIN employees m ON e.manager_id = m.id;

-- 查询同一班级的所有学生（排除自己）
SELECT a.name AS 学生1, b.name AS 学生2, a.grade AS 年级
FROM students a, students b
WHERE a.grade = b.grade AND a.id < b.id;

多表查询

SQL
-- 三表连接：学生、课程、成绩综合查询
SELECT
    s.name AS 学生姓名,
    s.grade AS 年级,
    c.course_name AS 课程名称,
    c.teacher AS 任课老师,
    sc.score AS 成绩,
    CASE
        WHEN sc.score >= 90 THEN 'A'
        WHEN sc.score >= 80 THEN 'B'
        WHEN sc.score >= 70 THEN 'C'
        WHEN sc.score >= 60 THEN 'D'
        ELSE 'F'
    END AS 等级
FROM students s
INNER JOIN scores sc ON s.id = sc.student_id
INNER JOIN courses c ON sc.course_id = c.id
WHERE s.grade = '高三'
ORDER BY s.name, c.course_name;
💡 JOIN 类型对比
连接类型说明
INNER JOIN只返回两边都匹配的行
LEFT JOIN返回左表所有行，右表匹配的返回，不匹配则为 NULL
RIGHT JOIN返回右表所有行，左表匹配的返回，不匹配则为 NULL
FULL JOIN返回两边所有行，不匹配则为 NULL
CROSS JOIN笛卡尔积，所有组合（慎用）
⚠️ 连接查询注意事项
  • 连接字段要有索引，否则大表连接会很慢
  • 避免大表之间的 CROSS JOIN，会产生海量数据
  • 多表连接时注意表的别名，提高可读性
  • LEFT JOIN 左边不要加 WHERE 条件到右表列上，会降级为 INNER JOIN

子查询

SQL
-- 标量子查询
SELECT name, (
    SELECT AVG(score) FROM scores WHERE student_id = s.id
) AS 平均分
FROM students s;

-- IN 子查询
SELECT * FROM students
WHERE id IN (
    SELECT DISTINCT student_id FROM scores WHERE score > 95
);

-- EXISTS 子查询
SELECT * FROM students s
WHERE EXISTS (
    SELECT 1 FROM scores sc
    WHERE sc.student_id = s.id AND sc.score > 95
);

-- FROM 子查询
SELECT name, 平均分
FROM (
    SELECT s.name, AVG(sc.score) AS 平均分
    FROM students s
    JOIN scores sc ON s.id = sc.student_id
    GROUP BY s.id
) AS student_avg
WHERE 平均分 > 90;

索引与事务

索引

SQL
-- 创建索引
CREATE INDEX idx_student_name ON students(name);
CREATE INDEX idx_score_student ON scores(student_id);

-- 唯一索引
CREATE UNIQUE INDEX idx_student_email ON students(email);

-- 组合索引（最左前缀原则）
CREATE INDEX idx_name_age ON students(name, age);

-- 查看索引
SHOW INDEX FROM students;

-- 删除索引
DROP INDEX idx_student_name ON students;

-- 查看执行计划
EXPLAIN SELECT * FROM students WHERE name = '张三';

事务

SQL
-- 开始事务
START TRANSACTION;
BEGIN;

-- 执行 SQL
UPDATE accounts SET balance = balance - 100 WHERE id = 1;
UPDATE accounts SET balance = balance + 100 WHERE id = 2;

-- 设置保存点
SAVEPOINT sp1;

-- 回滚到保存点
ROLLBACK TO sp1;

-- 提交事务
COMMIT;

-- 回滚事务
ROLLBACK;
💡 事务四大特性 (ACID)
  • 原子性 (Atomicity)：要么全成功，要么全失败
  • 一致性 (Consistency)：事务前后数据保持一致
  • 隔离性 (Isolation)：并发事务互不干扰
  • 持久性 (Durability)：提交后永久保存
