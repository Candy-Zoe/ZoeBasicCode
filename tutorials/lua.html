Lua 基础教程

Lua 简介

Lua 是一种轻量级、高效的嵌入式脚本语言，由巴西里约热内卢天主教大学开发。它以简洁、快速、可嵌入著称，广泛用于游戏开发（如魔兽世界、愤怒的小鸟）和嵌入式系统。

💡 Lua 特点
  • 轻量级：核心仅几百 KB
  • 高效：使用寄存器式虚拟机
  • 可嵌入：C API 丰富，易于集成
  • 基于表（Table）的数据结构
  • 元表（Metatable）实现面向对象
  • 协程（coroutine）支持

运算符

Lua 支持丰富的运算符，包括算术、关系和逻辑运算符。

Lua
-- 算术运算符
a, b = 10, 3
print("a = " .. a .. ", b = " .. b)
print("a + b = " .. (a + b))     -- 加法：13
print("a - b = " .. (a - b))     -- 减法：7
print("a * b = " .. (a * b))     -- 乘法：30
print("a / b = " .. (a / b))     -- 除法：3.333...
print("a % b = " .. (a % b))      -- 取模：1
print("a ^ b = " .. (a ^ b))      -- 幂：1000
print("-a = " .. (-a))           -- 取负：-10

-- 关系运算符（返回 true 或 false）
print("a > b  = " .. (a > b))    -- 大于：true
print("a < b  = " .. (a < b))    -- 小于：false
print("a == b = " .. (a == b))   -- 等于：false
print("a ~= b = " .. (a ~= b))   -- 不等于：true
print("a >= b = " .. (a >= b))   -- 大于等于：true
print("a <= b = " .. (a <= b))   -- 小于等于：false

-- 逻辑运算符
print("true and false = " .. (true and false))  -- and：false
print("true or false  = " .. (true or false))   -- or：true
print("not true       = " .. (not true))        -- not：false

-- Lua 的短路求值
x = x or 0  -- x 为 nil 时设为 0（类似默认值）
print("x or 0 = " .. x)

-- 三目运算符替代方案
local result = (age >= 18) and "成年" or "未成年"
print("年龄判断: " .. result)

-- 位运算符（Lua 5.3+）
print("5 & 3  = " .. (5 & 3))    -- 位与：1
print("5 | 3  = " .. (5 | 3))    -- 位或：7
print("5 ~ 3  = " .. (5 ~ 3))    -- 位异或：6
print("~5     = " .. (~5))        -- 位非
print("2 << 1 = " .. (2 << 1))   -- 左移：4
print("4 >> 1 = " .. (4 >> 1))   -- 右移：2
💡 运算符优先级（从高到低）
  • ^（幂运算，右结合）
  • not # - ~（一元运算符）
  • * / %（乘除取模）
  • + -（加减）
  • ..（字符串连接）
  • < > <= >= ~= ==（比较）
  • and
  • or
⚠️ 注意
  • Lua 中字符串连接用 ..，不是 +
  • 比较运算符两边不需要空格，但建议加上增加可读性
  • nil 和 false 在逻辑判断中为"假"，其他都为"真"

Table 是 Lua 中唯一的数据结构，可以表示数组、字典、对象、集合等。掌握 table 的操作是 Lua 编程的基础。

table 库函数

Lua
-- table.insert 在指定位置插入
fruits = {"苹果", "香蕉", "橙子"}
table.insert(fruits, "葡萄")           -- 末尾插入
print("插入后: " .. table.concat(fruits, ", "))

table.insert(fruits, 2, "西瓜")       -- 在位置 2 插入
print("在位置2插入后: " .. table.concat(fruits, ", "))

-- table.remove 删除并返回元素
removed = table.remove(fruits)        -- 删除末尾
print("删除: " .. removed)
print("删除后: " .. table.concat(fruits, ", "))

removed = table.remove(fruits, 1)     -- 删除位置 1
print("删除位置1: " .. removed)

-- table.sort 排序
nums = {3, 1, 4, 1, 5, 9, 2, 6}
table.sort(nums)
print("排序后: " .. table.concat(nums, ", "))

-- 自定义排序
fruits = {"香蕉", "苹果", "橙子"}
table.sort(fruits, function(a, b)
    return a > b  -- 降序排列
end)
print("降序: " .. table.concat(fruits, ", "))

-- table.concat 连接为字符串
fruits = {"苹果", "香蕉", "橙子"}
print("连接: " .. table.concat(fruits, ", "))
print("连接2-3: " .. table.concat(fruits, ", ", 2, 3))

数组操作

Lua
-- 获取数组长度（# 操作符）
arr = {10, 20, 30}
print("长度: " .. #arr)

-- 注意：# 只返回连续的数组部分长度
arr2 = {1, 2, nil, 4, 5}
print("#arr2 = " .. #arr2)  -- 可能不是 5，因为有 nil

-- 安全的数组长度获取方法
function array_size(arr)
    local count = 0
    for i in ipairs(arr) do
        count = i
    end
    return count
end
print("安全获取: " .. array_size(arr2))

-- 遍历数组
arr = {"a", "b", "c"}
for i, v in ipairs(arr) do
    print("arr[" .. i .. "] = " .. v)
end

-- 反向遍历
for i = #arr, 1, -1 do
    print("反向: arr[" .. i .. "] = " .. arr[i])
end

-- 数组切片
function slice(arr, start, ending)
    local result = {}
    for i = start, ending do
        table.insert(result, arr[i])
    end
    return result
end

nums = {1, 2, 3, 4, 5}
sliced = slice(nums, 2, 4)
print("切片[2-4]: " .. table.concat(sliced, ", "))

字典操作

Lua
-- 创建字典
person = {
    name = "张三",
    age = 25,
    city = "北京"
}

-- 遍历字典
for k, v in pairs(person) do
    print(k .. ": " .. v)
end

-- 获取所有键
function get_keys(dict)
    local keys = {}
    for k in pairs(dict) do
        table.insert(keys, k)
    end
    return keys
end

-- 获取所有值
function get_values(dict)
    local values = {}
    for _, v in pairs(dict) do
        table.insert(values, v)
    end
    return values
end

keys = get_keys(person)
print("所有键: " .. table.concat(keys, ", "))

values = get_values(person)
print("所有值: " .. table.concat(values, ", "))

-- 复制字典
function copy_dict(orig)
    local copy = {}
    for k, v in pairs(orig) do
        copy[k] = v
    end
    return copy
end

person2 = copy_dict(person)
person2.name = "李四"
print("原person.name: " .. person.name)
print("复制.name: " .. person2.name)
💡 table 库常用函数
  • table.insert(t, pos, val)：在 pos 位置插入 val，默认末尾
  • table.remove(t, pos)：删除 pos 位置元素，默认末尾
  • table.sort(t, comp)：排序，可选自定义比较函数
  • table.concat(t, sep, i, j)：连接为字符串
  • table.unpack：展开数组为多个返回值
  • table.pack：打包多个参数为数组
⚠️ table 重要注意事项
  • Lua 数组索引从 1 开始，不是 0！
  • # 操作符只对连续数组有效，有 nil 间隔时不准确
  • 删除元素用 t[k] = nil，而不是 table.remove
  • 字典遍历用 pairs，数组遍历用 ipairs

-- 多维数组（矩阵）
matrix = {
    {1, 2, 3},
    {4, 5, 6},
    {7, 8, 9}

-- 遍历矩阵
for i = 1, #matrix do
    for j = 1, #matrix[i] do
        io.write(matrix[i][j] .. " ")
    end
    print()
end

-- 创建 3x3 零矩阵
function new_matrix(rows, cols, init)
    init = init or 0
    local m = {}
    for i = 1, rows do
        m[i] = {}
        for j = 1, cols do
            m[i][j] = init
        end
    end
    return m
end

m = new_matrix(3, 3, 0)
print("零矩阵[2][2]: " .. m[2][2])  -- 0

函数

Lua
-- 基本函数
function add(a, b)
    return a + b
end
print("add(5, 3) = " .. add(5, 3))

-- 多返回值
function divide(a, b)
    return a / b, a % b
end
q, r = divide(10, 3)
print("商: " .. q .. ", 余: " .. r)

-- 可变参数
function sum(...)
    local total = 0
    local args = {...}
    for i = 1, #args do
        total = total + args[i]
    end
    return total
end
print("sum(1,2,3,4,5) = " .. sum(1, 2, 3, 4, 5))

-- 匿名函数
double = function(x)
    return x * 2
end
print("double(5) = " .. double(5))

-- 高阶函数
function apply(arr, func)
    local result = {}
    for i = 1, #arr do
        result[i] = func(arr[i])
    end
    return result
end

nums = {1, 2, 3}
result = apply(nums, function(x) return x ^ 2 end)
print(table.concat(result, ", "))

-- 闭包
function counter()
    local count = 0
    return function()
        count = count + 1
        return count
    end
end

c = counter()
print(c())  -- 1
print(c())  -- 2
print(c())  -- 3

if 条件语句

if 语句用于根据条件执行不同的代码块。

Lua
-- 基础 if 语句
score = 85

if score >= 90 then
    print("成绩等级：优秀")
end

-- if-else 二选一
if score >= 60 then
    print("及格了！")
else
    print("不及格，需要努力！")
end

-- if-elseif-else 多选一
if score >= 90 then
    print("优秀")
elseif score >= 80 then
    print("良好")
elseif score >= 60 then
    print("及格")
else
    print("不及格")
end

-- 条件组合
age = 20
income = 5000

if age >= 18 and income > 3000 then
    print("符合贷款条件")
end

-- 字符串比较
name = "Alice"
if name == "Alice" then
    print("欢迎，Alice！")
elseif name == "Bob" then
    print("欢迎，Bob！")
else
    print("访客，你好！")
end
💡 条件判断要点
  • then 和 end 是必须的
  • elseif 是一个词，不是 else if
  • 关系运算：== 等于，~= 不等于
  • 逻辑运算：and 与，or 或，not 非

循环

Lua 支持多种循环结构：while、repeat-until、for 循环。

Lua
-- while 循环：条件为真时重复执行
i = 1
while i <= 3 do
    print("while 循环: " .. i)
    i = i + 1
end

-- repeat...until：至少执行一次，直到条件为真
i = 1
repeat
    print("repeat 循环: " .. i)
    i = i + 1
until i > 3

-- for 数值循环：指定次数的迭代
-- 格式：for 变量 = 初值, 终值, 步长 do ... end
for i = 1, 3 do
    print("for 循环: " .. i)
end

-- 步长为 2
for i = 1, 5, 2 do
    print("步长2: " .. i)
end

-- 倒序
for i = 3, 1, -1 do
    print("倒序: " .. i)
end

break 与 return

Lua
-- break：退出循环
print("break 示例：")
for i = 1, 10 do
    if i > 5 then
        break
    end
    print("  i = " .. i)
end

-- return：函数中返回，模块中提前退出
function find_element(arr, target)
    for i, v in ipairs(arr) do
        if v == target then
            return i  -- 找到就返回索引
        end
    end
    return nil  -- 没找到返回 nil
end

result = find_element({1, 2, 3, 4, 5}, 3)
print("元素 3 的索引: " .. result)
⚠️ 循环注意事项
  • Lua 数组索引从 1 开始，不是 0！
  • ipairs 只遍历连续的数字索引，遇到 nil 停止
  • pairs 遍历所有键值对，顺序不确定
  • 循环中用 break 退出，用 return 在函数中退出

元表与面向对象

元表

-- 元表：修改表的行为
t1 = {1, 2, 3}
t2 = {4, 5, 6}

mt = {
    __add = function(a, b)
        local result = {}
        for i = 1, #a do
            result[i] = a[i] + b[i]
        end
        return result
    end
}

setmetatable(t1, mt)
setmetatable(t2, mt)

t3 = t1 + t2
print("表相加: " .. table.concat(t3, ", "))

-- 常用元方法：
-- __add 加  __sub 减  __mul 乘  __div 除

-- __index 访问不存在的键  __newindex 赋值
-- __tostring 转字符串  __call 调用表

面向对象

Lua
-- 原型模式（基于表的 OOP）
Account = { balance = 0 }

function Account:new(o)
    o = o or {}
    setmetatable(o, self)
    self.__index = self
    return o
end

function Account:deposit(v)
    self.balance = self.balance + v
end

function Account:withdraw(v)
    if v <= self.balance then
        self.balance = self.balance - v
    end
end

a = Account:new{balance = 100}
a:deposit(50)
print("存款后余额: " .. a.balance)
a:withdraw(30)
print("取款后余额: " .. a.balance)

-- 继承
SpecialAccount = Account:new()

function SpecialAccount:withdraw(v)
    if v - self.balance >= 0 then
        self.balance = self.balance - v
    end
end

s = SpecialAccount:new{balance = 100}
s:withdraw(200)  -- 可以透支
print("特殊账户余额: " .. s.balance)

错误处理

Lua
-- error 抛出错误
function divide(a, b)
    if b == 0 then
        error("除数不能为零")
    end
    return a / b
end

-- pcall 保护调用
ok, result = pcall(divide, 10, 0)
if ok then
    print("成功: " .. result)
else
    print("错误: " .. result)
end

ok, result = pcall(divide, 10, 2)
if ok then
    print("成功: " .. result)
end

-- xpcall 带错误处理
function errorHandler(err)
    return "捕获到错误: " .. err
end

ok, result = xpcall(function() return divide(5, 0) end, errorHandler)
print(result)

-- assert 断言
n = 5
assert(n > 0, "n 必须大于 0")
print("assert 通过")
