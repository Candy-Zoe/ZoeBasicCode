Lua 基础教程

Lua 简介

Lua 是一种轻量级、高效的嵌入式脚本语言，由巴西里约热内卢天主教大学开发。它以简洁、快速、可嵌入著称，广泛用于游戏开发（如魔兽世界、愤怒的小鸟）和嵌入式系统。

💡 Lua 特点
  • 轻量级：核心仅几百 KB
  • 高效：使用寄存器式虚拟机
  • 可嵌入：C API 丰富，易于集成
  • 基于表（Table）的数据结构
  • 元表（Metatable）实现面向对象
  • 协程（coroutine）支持

Table 表

Table 是 Lua 中唯一的数据结构，可以表示数组、字典、对象、集合等。

Lua
-- 数组（数字索引，从 1 开始！）
fruits = {"苹果", "香蕉", "橙子"}
print(fruits[1])      -- 第一个元素（不是 0！）
print(#fruits)        -- 长度

-- 遍历数组
for i, v in ipairs(fruits) do
    print(i .. ". " .. v)
end

-- 字典（字符串索引）
person = {
    name = "张三",
    age = 25,
    city = "北京"
}
print(person.name)
print(person["age"])

-- 遍历字典
for k, v in pairs(person) do
    print(k .. ": " .. v)
end

-- 混合表
mixed = {
    10,
    20,
    name = "test",
    value = 100
}
print(mixed[1])
print(mixed.name)

-- table 库
table.insert(fruits, "葡萄")     -- 添加
table.remove(fruits, 1)          -- 删除
table.sort(fruits)               -- 排序
table.concat(fruits, ", ")       -- 连接

-- 多维表
matrix = {
    {1, 2, 3},
    {4, 5, 6},
    {7, 8, 9}
}
print(matrix[2][3])  -- 6
⚠️ 注意
  • Lua 数组索引从 1 开始，不是 0！
  • ipairs 只遍历连续的数字索引
  • pairs 遍历所有键值对，顺序不确定
  • # 操作符只对数组部分有效

函数

Lua
-- 基本函数
function add(a, b)
    return a + b
end
print("add(5, 3) = " .. add(5, 3))

-- 多返回值
function divide(a, b)
    return a / b, a % b
end
q, r = divide(10, 3)
print("商: " .. q .. ", 余: " .. r)

-- 可变参数
function sum(...)
    local total = 0
    local args = {...}
    for i = 1, #args do
        total = total + args[i]
    end
    return total
end
print("sum(1,2,3,4,5) = " .. sum(1, 2, 3, 4, 5))

-- 匿名函数
double = function(x)
    return x * 2
end
print("double(5) = " .. double(5))

-- 高阶函数
function apply(arr, func)
    local result = {}
    for i = 1, #arr do
        result[i] = func(arr[i])
    end
    return result
end

nums = {1, 2, 3}
result = apply(nums, function(x) return x ^ 2 end)
print(table.concat(result, ", "))

-- 闭包
function counter()
    local count = 0
    return function()
        count = count + 1
        return count
    end
end

c = counter()
print(c())  -- 1
print(c())  -- 2
print(c())  -- 3

元表与面向对象

元表

Lua
-- 元表：修改表的行为
t1 = {1, 2, 3}
t2 = {4, 5, 6}

mt = {
    __add = function(a, b)
        local result = {}
        for i = 1, #a do
            result[i] = a[i] + b[i]
        end
        return result
    end
}

setmetatable(t1, mt)
setmetatable(t2, mt)

t3 = t1 + t2
print("表相加: " .. table.concat(t3, ", "))

-- 常用元方法：
-- __add 加  __sub 减  __mul 乘  __div 除
-- __eq 等于  __lt 小于  __le 小于等于
-- __index 访问不存在的键  __newindex 赋值
-- __tostring 转字符串  __call 调用表

面向对象

Lua
-- 原型模式（基于表的 OOP）
Account = { balance = 0 }

function Account:new(o)
    o = o or {}
    setmetatable(o, self)
    self.__index = self
    return o
end

function Account:deposit(v)
    self.balance = self.balance + v
end

function Account:withdraw(v)
    if v <= self.balance then
        self.balance = self.balance - v
    end
end

a = Account:new{balance = 100}
a:deposit(50)
print("存款后余额: " .. a.balance)
a:withdraw(30)
print("取款后余额: " .. a.balance)

-- 继承
SpecialAccount = Account:new()

function SpecialAccount:withdraw(v)
    if v - self.balance >= 0 then
        self.balance = self.balance - v
    end
end

s = SpecialAccount:new{balance = 100}
s:withdraw(200)  -- 可以透支
print("特殊账户余额: " .. s.balance)

错误处理

Lua
-- error 抛出错误
function divide(a, b)
    if b == 0 then
        error("除数不能为零")
    end
    return a / b
end

-- pcall 保护调用
ok, result = pcall(divide, 10, 0)
if ok then
    print("成功: " .. result)
else
    print("错误: " .. result)
end

ok, result = pcall(divide, 10, 2)
if ok then
    print("成功: " .. result)
end

-- xpcall 带错误处理
function errorHandler(err)
    return "捕获到错误: " .. err
end

ok, result = xpcall(function() return divide(5, 0) end, errorHandler)
print(result)

-- assert 断言
n = 5
assert(n > 0, "n 必须大于 0")
print("assert 通过")
