Go 语言基础语法教程

Go 简介

Go（又称 Golang）是 Google 开发的一种静态类型、编译型语言，以简洁、高效、并发支持好著称。

💡 Go 语言特点
  • 语法简洁，只有 25 个关键字
  • 内置并发支持（goroutine + channel）
  • 垃圾回收（自动内存管理）
  • 编译速度快
  • 静态链接二进制文件，部署简单

    fmt.Println("Hello, World!")

函数

基本函数

Go 
package main

import "fmt"

// 多个返回值
func calculate(a, b int) (int, int) {
    return a + b, a * b
}

// 命名返回值
func divide(a, b int) (quotient, remainder int) {
    quotient = a / b
    remainder = a % b
    return
}

func main() {
    sum, product := calculate(10, 3)
    fmt.Printf("和: %d, 积: %d\n", sum, product)
    
    q, r := divide(10, 3)
    fmt.Printf("商: %d, 余数: %d\n", q, r)
}
和: 13, 积: 30
商: 3, 余数: 1

可变参数

Go 
func sumAll(nums ...int) int {
    total := 0
    for _, n := range nums {
        total += n
    }
    return total
}

func main() {
    fmt.Println("sum:", sumAll(1, 2, 3))
    fmt.Println("sum:", sumAll(1, 2, 3, 4, 5))
}
sum: 6
sum: 15

结构体

定义和使用

Go 
package main

import "fmt"

type Person struct {
    Name string
    Age  int
}

// 方法
func (p Person) Greet() {
    fmt.Printf("你好，我是%s\n", p.Name)
}

func main() {
    p := Person{Name: "张三", Age: 25}
    p.Greet()
    fmt.Printf("年龄: %d\n", p.Age)
}
你好，我是张三
年龄: 25

接口

接口定义

Go 
package main

import "fmt"

type Shape interface {
    Area() float64
}

type Rectangle struct {
    Width, Height float64
}

func (r Rectangle) Area() float64 {
    return r.Width * r.Height
}

func main() {
    var s Shape = Rectangle{Width: 4, Height: 5}
    fmt.Printf("面积: %.2f\n", s.Area())
}
💡 Go 接口特点 Go 的接口是隐式实现，不需要显式声明 implements。

并发编程

goroutine

Go 
package main

import (
    "fmt"
    "time"
)

func say(s string) {
    for i := 0; i < 3; i++ {
        fmt.Println(s)
    }
}

func main() {
    go say("异步执行")
    say("同步执行")
    time.Sleep(time.Second)  // 等待 goroutine 完成
}

channel 通道

Go 
func main() {
    ch := make(chan string)
    
    go func() {
        ch <- "数据1"
        ch <- "数据2"
        close(ch)
    }()
    
    for msg := range ch {
        fmt.Println("收到:", msg)
    }
}
