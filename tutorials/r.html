R 语言简介

R 是一种用于统计计算和图形展示的编程语言和环境，广泛用于数据分析、统计建模和可视化。由新西兰奥克兰大学开发，现已成为数据科学领域的重要工具。

💡 R 语言特点
  • 强大的统计和数据分析能力
  • 出色的绘图和可视化能力
  • 丰富的包生态系统（CRAN）
  • 向量化操作，高效处理数据
  • 函数式编程特性
  • 活跃的社区和大量学习资源

向量 Vector

向量是 R 中最基本的数据结构，所有元素必须是同一类型。

R
# 创建向量
v <- c(1, 2, 3, 4, 5)
print(v)

# 序列
v2 <- 1:10
print(v2)

v3 <- seq(1, 10, by = 2)
print(v3)

v4 <- rep("A", 5)
print(v4)

# 向量运算（向量化）
a <- c(1, 2, 3)
b <- c(4, 5, 6)
cat("a + b =", a + b, "\n")
cat("a * b =", a * b, "\n")
cat("a ^ 2 =", a ^ 2, "\n")

# 常用函数
nums <- c(5, 2, 8, 1, 9, 3)
cat("长度:", length(nums), "\n")
cat("总和:", sum(nums), "\n")
cat("均值:", mean(nums), "\n")
cat("中位数:", median(nums), "\n")
cat("方差:", var(nums), "\n")
cat("标准差:", sd(nums), "\n")
cat("最大值:", max(nums), "\n")
cat("最小值:", min(nums), "\n")
cat("排序:", sort(nums), "\n")
cat("反转:", rev(nums), "\n")

# 索引（从 1 开始！）
v <- c("a", "b", "c", "d", "e")
cat("v[1] =", v[1], "\n")
cat("v[2:4] =", v[2:4], "\n")
cat("v[c(1,3,5)] =", v[c(1,3,5)], "\n")
cat("v[-1] =", v[-1], "\n")  # 删除第一个

# 逻辑索引
nums <- 1:10
cat("大于5的:", nums[nums > 5], "\n")

# 命名向量
scores <- c(语文=95, 数学=88, 英语=92)
cat("数学成绩:", scores["数学"], "\n")

函数

R
# 基本函数
add <- function(a, b) {
    return(a + b)
}
cat("add(3, 5) =", add(3, 5), "\n")

# 默认参数
greet <- function(name = "世界") {
    cat("你好,", name, "!\n")
}
greet()
greet("张三")

# 可变参数
my_sum <- function(...) {
    args <- list(...)
    return(sum(unlist(args)))
}
cat("my_sum(1,2,3,4,5) =", my_sum(1,2,3,4,5), "\n")

# 函数作为参数
apply_func <- function(arr, func) {
    return(sapply(arr, func))
}
result <- apply_func(1:5, function(x) x^2)
cat("平方:", result, "\n")

# 递归
factorial <- function(n) {
    if (n <= 1) {
        return(1)
    }
    return(n * factorial(n - 1))
}
cat("5! =", factorial(5), "\n")

流程控制

R
# if-else
score <- 85
if (score >= 90) {
    cat("优秀\n")
} else if (score >= 80) {
    cat("良好\n")
} else if (score >= 60) {
    cat("及格\n")
} else {
    cat("不及格\n")
}

# ifelse（向量化）
scores <- c(95, 82, 75, 60, 55)
grades <- ifelse(scores >= 90, "优秀",
          ifelse(scores >= 60, "及格", "不及格"))
cat("成绩分级:", grades, "\n")

# for 循环
cat("for 循环:\n")
for (i in 1:3) {
    cat("  ", i, "\n")
}

# 遍历向量
fruits <- c("苹果", "香蕉", "橙子")
cat("遍历水果:\n")
for (f in fruits) {
    cat("  -", f, "\n")
}

# while 循环
cat("while 循环:\n")
i <- 1
while (i <= 3) {
    cat("  ", i, "\n")
    i <- i + 1
}

# repeat 循环
cat("repeat 循环:\n")
i <- 1
repeat {
    cat("  ", i, "\n")
    i <- i + 1
    if (i > 3) break
}

# break 和 next
cat("break/next:\n")
for (i in 1:5) {
    if (i == 3) next  # 跳过
    if (i == 5) break # 退出
    cat("  ", i, "\n")
}

# switch
day <- "周一"
result <- switch(day,
    "周一" = "工作日",
    "周二" = "工作日",
    "周三" = "工作日",
    "周四" = "工作日",
    "周五" = "工作日",
    "周六" = "周末",
    "周日" = "周末"
)
cat("周一:", result, "\n")

apply 族函数

apply 族函数是 R 的精髓，用于对数据应用函数，避免显式循环。

R
# apply - 矩阵/数组
m <- matrix(1:9, nrow=3)
cat("行求和:\n")
print(apply(m, 1, sum))
cat("列求平均:\n")
print(apply(m, 2, mean))

# lapply - 列表/向量，返回列表
nums <- list(a=1:3, b=4:6, c=7:9)
cat("lapply 求和:\n")
print(lapply(nums, sum))

# sapply - 简化结果
cat("sapply 求和:\n")
print(sapply(nums, sum))

# tapply - 分组应用
df <- data.frame(
    name = c("A","B","C","D","E"),
    group = c("甲","乙","甲","乙","甲"),
    score = c(90, 85, 92, 88, 95)
)
cat("分组平均值:\n")
print(tapply(df$score, df$group, mean))

# vapply - 指定返回类型
cat("vapply 长度:\n")
print(vapply(nums, length, integer(1)))

# mapply - 多参数
cat("mapply:\n")
print(mapply(function(a, b) a + b, c(1,2,3), c(4,5,6)))

统计分析

R
# 生成随机数据
set.seed(42)  # 设置随机种子
data <- rnorm(100, mean=0, sd=1)  # 正态分布

# 描述性统计
cat("长度:", length(data), "\n")
cat("均值:", mean(data), "\n")
cat("中位数:", median(data), "\n")
cat("方差:", var(data), "\n")
cat("标准差:", sd(data), "\n")
cat("最小值:", min(data), "\n")
cat("最大值:", max(data), "\n")
cat("分位数:\n")
print(quantile(data))
cat("摘要:\n")
print(summary(data))

# 相关性
x <- 1:10
y <- 2*x + rnorm(10)
cat("相关系数:", cor(x, y), "\n")

# 线性回归
model <- lm(y ~ x)
cat("回归模型:\n")
print(summary(model))

# 假设检验
t.test(data, mu=0)  # t 检验

# 常用分布函数
# rnorm  rbinom  runif  rexp  # 生成随机数
# dnorm  dbinom  dunif  dexp  # 概率密度
# pnorm  pbinom  punif  pexp  # 累积概率
# qnorm  qbinom  qunif  qexp  # 分位数
