R 语言简介

R 是一种用于统计计算和图形展示的编程语言和环境，广泛用于数据分析、统计建模和可视化。由新西兰奥克兰大学开发，现已成为数据科学领域的重要工具。

💡 R 语言特点
  • 强大的统计和数据分析能力
  • 出色的绘图和可视化能力
  • 丰富的包生态系统（CRAN）
  • 向量化操作，高效处理数据
  • 函数式编程特性
  • 活跃的社区和大量学习资源

向量 Vector

向量是 R 中最基本的数据结构，所有元素必须是同一类型。R 的很多操作都是向量化的。

R
# 创建向量
v <- c(1, 2, 3, 4, 5)
print(v)

# 序列
v2 <- 1:10
print(v2)

v3 <- seq(1, 10, by = 2)  # 步长为 2
print(v3)

v4 <- rep("A", 5)  # 重复 A 5 次
print(v4)

# 向量运算（向量化！R 的精髓）
a <- c(1, 2, 3)
b <- c(4, 5, 6)
cat("a + b =", a + b, "\n")
cat("a * b =", a * b, "\n")
cat("a ^ 2 =", a ^ 2, "\n")
cat("a + 10 =", a + 10, "\n")

# 向量筛选
nums <- 1:10
cat("大于5的:", nums[nums > 5], "\n")
cat("偶数:", nums[nums %% 2 == 0], "\n")

向量运算函数

R
# 常用统计函数
nums <- c(5, 2, 8, 1, 9, 3)
cat("长度:", length(nums), "\n")
cat("总和:", sum(nums), "\n")
cat("均值:", mean(nums), "\n")
cat("中位数:", median(nums), "\n")
cat("方差:", var(nums), "\n")
cat("标准差:", sd(nums), "\n")
cat("最大值:", max(nums), "\n")
cat("最小值:", min(nums), "\n")
cat("排序:", sort(nums), "\n")
cat("反转:", rev(nums), "\n")
cat("乘积:", prod(nums), "\n")

# 累计函数
cat("累计和:", cumsum(nums), "\n")
cat("累计积:", cumprod(nums), "\n")
cat("滞后:", lag(nums), "\n")
cat("差分:", diff(nums), "\n")

# 索引（从 1 开始！）
v <- c("a", "b", "c", "d", "e")
cat("v[1] =", v[1], "\n")
cat("v[2:4] =", v[2:4], "\n")
cat("v[c(1,3,5)] =", v[c(1,3,5)], "\n")
cat("v[-1] =", v[-1], "\n")  # 删除第一个

# 命名向量
scores <- c(语文=95, 数学=88, 英语=92)
cat("数学成绩:", scores["数学"], "\n")
cat("所有名称:", names(scores), "\n")
💡 向量化的好处
  • 代码简洁，不需要显式循环
  • 执行效率高，R 内部做了优化
  • 大多数函数都支持向量化操作

矩阵是二维数据结构，所有元素必须是同一类型。

矩阵运算

R
# 创建两个 2x2 矩阵
m1 <- matrix(1:4, nrow=2)
m2 <- matrix(5:8, nrow=2)

# 元素级运算
print(m1 + m2)    # 加法
print(m1 - m2)    # 减法
print(m1 * m2)    # 乘法（元素级）
print(m1 / m2)    # 除法（元素级）

# 矩阵乘法（%*%）
result <- m1 %*% m2
print(result)

# 转置
print(t(m1))

# 行列操作
m <- matrix(1:9, nrow=3)
cat("行和:", apply(m, 1, sum), "\n")      # 每行求和
cat("列和:", apply(m, 2, sum), "\n")      # 每列求和
cat("行均值:", apply(m, 1, mean), "\n")  # 每行均值
cat("列均值:", apply(m, 2, mean), "\n")  # 每列均值

# 对角矩阵和单位矩阵
diag(3)           # 3x3 单位矩阵
diag(c(1,2,3))    # 对角矩阵

# 矩阵索引
cat("m[1,1] =", m[1,1], "\n")
cat("m[2,] =", m[2,], "\n")
cat("m[,3] =", m[,3], "\n")
cat("m[1:2, 2:3] =\n")
print(m[1:2, 2:3])

数据框 Data Frame

数据框是 R 中最常用的数据结构，类似于 Excel 表格，每列可以是不同类型。

R
# 创建数据框
df <- data.frame(
    姓名 = c("张三", "李四", "王五", "赵六"),
    年龄 = c(25, 30, 28, 35),
    城市 = c("北京", "上海", "广州", "深圳"),
    成绩 = c(95, 88, 92, 85),
    stringsAsFactors = FALSE  # 避免字符串自动转为因子
)
print(df)

# 查看基本信息
cat("行列数:", dim(df), "\n")
cat("行数:", nrow(df), "\n")
cat("列数:", ncol(df), "\n")
cat("列名:", colnames(df), "\n")
cat("结构:\n")
str(df)

# 数据框索引
cat("姓名列:", df$姓名, "\n")
cat("第二行:", as.character(df[2, ]), "\n")
cat("姓名和成绩列前2行:\n")
print(df[1:2, c("姓名", "成绩")])

数据框操作

R
# 创建示例数据框
df <- data.frame(
    姓名 = c("张三", "李四", "王五", "赵六", "钱七"),
    年龄 = c(25, 30, 28, 35, 22),
    城市 = c("北京", "上海", "广州", "深圳", "北京"),
    成绩 = c(95, 88, 92, 85, 78),
    stringsAsFactors = FALSE
)

# 条件筛选
cat("成绩大于90的行:\n")
print(df[df$成绩 > 90, ])

cat("\n北京的学生:\n")
print(df[df$城市 == "北京", ])

cat("\n成绩最高的学生:\n")
print(df[which.max(df$成绩), ])

# 添加新列
df$等级 <- ifelse(df$成绩 >= 90, "优秀",
            ifelse(df$成绩 >= 80, "良好", "一般"))
print(df)

# 添加计算列
df$成绩加分后 <- df$成绩 + 5

# 删除列
df$临时列 <- NULL

# 排序
df_sorted <- df[order(df$成绩, decreasing = TRUE), ]
cat("\n按成绩降序:\n")
print(df_sorted)

# 统计计算
cat("\n各城市平均成绩:\n")
print(tapply(df$成绩, df$城市, mean))

# 数据框合并
df1 <- data.frame(ID = 1:3, 姓名 = c("甲", "乙", "丙"))
df2 <- data.frame(ID = 1:3, 成绩 = c(90, 85, 92))
merged <- merge(df1, df2, by = "ID")
cat("\n合并后的数据框:\n")
print(merged)

基础绘图

R 最突出的特点之一就是其强大的绘图能力。使用基础绘图系统可以快速创建各种统计图形。

plot() 散点图

R
# 基础散点图
x <- 1:50
y <- x + rnorm(50, 0, 5)  # 添加随机噪声
plot(x, y)

# 带参数的散点图
plot(x, y,
     main = "散点图示例",      # 标题
     xlab = "X 轴",            # X 轴标签
     ylab = "Y 轴",            # Y 轴标签
     col = "blue",             # 颜色
     pch = 16,                 # 点型（16 是实心圆）
     cex = 1.5)                # 点的大小

# 添加参考线
abline(h = mean(y), col = "red", lty = 2)  # 水平线
abline(v = 25, col = "green", lty = 2)     # 垂直线
abline(lm(y ~ x), col = "orange")           # 回归线

函数

R
# 基本函数
add <- function(a, b) {
    return(a + b)
}
cat("add(3, 5) =", add(3, 5), "\n")

# 默认参数
greet <- function(name = "世界") {
    cat("你好,", name, "!\n")
}
greet()
greet("张三")

# 可变参数
my_sum <- function(...) {
    args <- list(...)
    return(sum(unlist(args)))
}
cat("my_sum(1,2,3,4,5) =", my_sum(1,2,3,4,5), "\n")

# 函数作为参数
apply_func <- function(arr, func) {
    return(sapply(arr, func))
}
result <- apply_func(1:5, function(x) x^2)
cat("平方:", result, "\n")

# 递归
factorial <- function(n) {
    if (n <= 1) {
        return(1)
    }
    return(n * factorial(n - 1))
}
cat("5! =", factorial(5), "\n")

流程控制

# if-else
score <- 85

    cat("优秀\n")
} else if (score >= 80) {
    cat("良好\n")
} else if (score >= 60) {
    cat("及格\n")

    cat("不及格\n")

          ifelse(scores >= 60, "及格", "不及格"))

# for 循环

for (i in 1:3) {
    cat("  ", i, "\n")
}

# 遍历向量
fruits <- c("苹果", "香蕉", "橙子")
cat("遍历水果:\n")
for (f in fruits) {
    cat("  -", f, "\n")
}

# while 循环
cat("while 循环:\n")
i <- 1
while (i <= 3) {
    cat("  ", i, "\n")
    i <- i + 1
}

# repeat 循环

    cat("  ", i, "\n")

    if (i == 3) next  # 跳过
    if (i == 5) break # 退出
    cat("  ", i, "\n")
}

    "周三" = "工作日",
    "周四" = "工作日",
    "周五" = "工作日",
    "周六" = "周末",

)

apply 族函数

apply 族函数是 R 的精髓，用于对数据应用函数，避免显式循环。

R
# apply - 矩阵/数组
m <- matrix(1:9, nrow=3)
cat("行求和:\n")
print(apply(m, 1, sum))
cat("列求平均:\n")
print(apply(m, 2, mean))

# lapply - 列表/向量，返回列表
nums <- list(a=1:3, b=4:6, c=7:9)
cat("lapply 求和:\n")
print(lapply(nums, sum))

# sapply - 简化结果
cat("sapply 求和:\n")
print(sapply(nums, sum))

# tapply - 分组应用
df <- data.frame(
    name = c("A","B","C","D","E"),
    group = c("甲","乙","甲","乙","甲"),
    score = c(90, 85, 92, 88, 95)
)
cat("分组平均值:\n")
print(tapply(df$score, df$group, mean))

# vapply - 指定返回类型
cat("vapply 长度:\n")
print(vapply(nums, length, integer(1)))

# mapply - 多参数
cat("mapply:\n")
print(mapply(function(a, b) a + b, c(1,2,3), c(4,5,6)))

R
# 生成随机数据
set.seed(42)  # 设置随机种子
data <- rnorm(100, mean=0, sd=1)  # 正态分布

# 描述性统计
cat("长度:", length(data), "\n")
cat("均值:", mean(data), "\n")
cat("中位数:", median(data), "\n")
cat("方差:", var(data), "\n")
cat("标准差:", sd(data), "\n")
cat("最小值:", min(data), "\n")
cat("最大值:", max(data), "\n")
cat("分位数:\n")
print(quantile(data))
cat("摘要:\n")
print(summary(data))

# 相关性
x <- 1:10
y <- 2*x + rnorm(10)
cat("相关系数:", cor(x, y), "\n")

# 线性回归
model <- lm(y ~ x)
