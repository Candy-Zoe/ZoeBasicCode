C++ 基础语法教程

C++ 简介

C++ 是 C 语言的扩展，增加了面向对象、泛型编程等特性，是目前最流行的系统级编程语言之一。

💡 C++ 与 C 的区别
  • 支持面向对象编程（类、继承、多态）
  • 标准模板库（STL）
  • 命名空间（namespace）
  • 引用（reference）类型
  • 异常处理

第一个程序

C++ 
#include <iostream>
using namespace std;

int main() {
    cout << "Hello, World!" << endl;
    return 0;
}
⚠️ 编译运行 g++ -std=c++17 hello.cpp -o hello && ./hello

变量和数据类型

基本类型

C++ 
int age = 25;                 // 整型
float height = 1.75f;           // 单精度
double weight = 65.5;          // 双精度（常用）
char grade = 'A';              // 字符
bool isStudent = true;         // 布尔
string name = "张三";          // 字符串（C++11）

cout << "姓名: " << name << endl;
cout << "年龄: " << age << endl;

auto 类型推断

C++ 
auto x = 10;           // int
auto y = 3.14;         // double
auto z = "hello"s;     // string (C++14)

cout << "x 类型: " << typeid(x).name() << endl;

vector 动态数组

C++ 
#include <vector>
using namespace std;

vector<int> nums = {1, 2, 3, 4, 5};
nums.push_back(6);           // 添加元素
cout << "大小: " << nums.size() << endl;
cout << "第一个: " << nums.front() << endl;
cout << "最后一个: " << nums.back() << endl;

面向对象

类定义

C++ 
class Person {
private:
    string name;
    int age;

public:
    Person(string n, int a) : name(n), age(a) {}
    
    void greet() {
        cout << "你好，我是" << name << endl;
    }
};

int main() {
    Person p("张三", 25);
    p.greet();
    return 0;
}
你好，我是张三

继承

C++ 
class Animal {
protected:
    string name;
public:
    Animal(string n) : name(n) {}
    virtual void speak() = 0;  // 纯虚函数
};

class Dog : public Animal {
public:
    Dog(string n) : Animal(n) {}
    void speak() override {
        cout << name << " 汪汪叫" << endl;
    }
};

int main() {
    Dog d("旺财");
    d.speak();
    return 0;
}

STL 容器

map 字典

C++ 
#include <map>
using namespace std;

map<string, int> scores;
scores["张三"] = 95;
scores["李四"] = 85;

for (auto& p : scores) {
    cout << p.first << ": " << p.second << endl;
}
李四: 85
张三: 95

set 集合

C++ 
#include <set>
using namespace std;

set<int> s = {3, 1, 4, 1, 5, 9};
cout << "集合(自动排序去重): ";
for (int n : s) cout << n << " ";
cout << endl;

函数和模板

函数模板

C++ 
template<typename T>
T maxValue(T a, T b) {
    return a > b ? a : b;
}

int main() {
    cout << maxValue(3, 5) << endl;
    cout << maxValue(3.14, 2.71) << endl;
    return 0;
}

Lambda 表达式

C++ 
#include <algorithm>
#include <vector>
using namespace std;

int main() {
    vector<int> nums = {3, 1, 4, 1, 5};
    
    auto square = [](int x) { return x * x; };
    cout << "5的平方: " << square(5) << endl;
    
    sort(nums.begin(), nums.end(), [](int a, int b) { return a > b; });
    cout << "降序排序: ";
    for (int n : nums) cout << n << " ";
    return 0;
}
